### 第 4 步：创建规则扫描任务

创建文件 `python_backend/app/rules/tasks.py`：

```python
"""Celery tasks for rules engine."""

from datetime import datetime, timezone
from uuid import uuid4

from app.tasks.celery_app import celery_app
from app.core.logging import get_logger
from app.db import get_connection
from app.graphs.supervisor import run_supervisor_graph_async
from app.graphs.tools import SupervisorTools, TaskServiceTaskTool, ApprovalServiceApprovalTool, TaskActionAuditLogTool
from app.graphs.rules_adapter import SupervisorRulesAdapter
from app.rules.engine import RulesEngine

logger = get_logger(__name__)


@celery_app.task(name="rules.scan_and_execute")
def scan_and_execute_rules():
    """扫描业务数据，执行规则引擎，自动创建任务。
    
    这个任务会：
    1. 从数据库读取库存、设备数据
    2. 执行规则引擎识别异常
    3. 通过 Supervisor Agent 创建任务
    """
    import asyncio
    
    async def _run():
        logger.info("Starting rules scan and execution")
        
        try:
            async with get_connection() as conn:
                # 1. 读取业务数据
                from app.api.inventory import get_chemicals_from_db, get_equipment_from_db
                chemicals = await get_chemicals_from_db(conn)
                equipment = await get_equipment_from_db(conn)
                
                logger.info(f"Loaded {len(chemicals)} chemicals, {len(equipment)} equipment")
                
                # 2. 执行规则引擎
                rules_engine = RulesEngine()
                events = rules_engine.inspect({
                    "chemicals": chemicals,
                    "equipment": equipment,
                })
                
                logger.info(f"Rules engine found {len(events)} events")
                
                # 3. 为每个事件运行 Supervisor Agent
                created_tasks = []
                for event in events:
                    try:
                        # 创建 Supervisor Tools
                        task_tool = TaskServiceTaskTool(get_connection)
                        approval_tool = ApprovalServiceApprovalTool(get_connection)
                        audit_log_tool = TaskActionAuditLogTool(get_connection)
                        rules_adapter = SupervisorRulesAdapter(rules_engine, task_tool=task_tool)
                        
                        tools = SupervisorTools(
                            task_tool=task_tool,
                            approval_tool=approval_tool,
                            audit_log_tool=audit_log_tool,
                            rules_adapter=rules_adapter,
                        )
                        
                        # 运行 Supervisor Graph
                        actor = {
                            "id": "system",
                            "name": "Rules Scanner",
                            "type": "system"
                        }
                        
                        state = await run_supervisor_graph_async(
                            event=event,
                            actor=actor,
                            tools=tools
                        )
                        
                        task_id = state.get("output", {}).get("taskId")
                        if task_id:
                            created_tasks.append(task_id)
                            logger.info(f"Created task {task_id} for event {event.get('type')}")
                    
                    except Exception as e:
                        logger.error(f"Failed to process event {event.get('id')}: {e}", exc_info=True)
                        continue
                
                logger.info(f"Rules scan completed: {len(created_tasks)} tasks created")
                
                return {
                    "events_found": len(events),
                    "tasks_created": len(created_tasks),
                    "task_ids": created_tasks,
                }
        
        except Exception as e:
            logger.error(f"Rules scan failed: {e}", exc_info=True)
            raise
    
    return asyncio.run(_run())
```

### 第 5 步：创建报告生成任务

创建文件 `python_backend/app/reports/scheduled_tasks.py`：

```python
"""Scheduled report generation tasks."""

from datetime import date, timedelta
from uuid import uuid4

from app.tasks.celery_app import celery_app
from app.core.logging import get_logger
from app.reports.dispatcher import submit_daily_report, submit_weekly_report

logger = get_logger(__name__)


@celery_app.task(name="reports.generate_daily")
def generate_daily_report_scheduled():
    """每天自动生成日报。"""
    import asyncio
    
    async def _run():
        today = date.today()
        yesterday = today - timedelta(days=1)
        
        logger.info(f"Generating scheduled daily report for {yesterday}")
        
        result = await submit_daily_report(
            target_date=yesterday.isoformat(),
            operator="system",
            run_id=str(uuid4())
        )
        
        logger.info(f"Daily report generated: {result}")
        return result
    
    return asyncio.run(_run())


@celery_app.task(name="reports.generate_weekly")
def generate_weekly_report_scheduled():
    """每周自动生成周报。"""
    import asyncio
    
    async def _run():
        today = date.today()
        # 上周一到上周日
        days_since_monday = today.weekday()
        last_monday = today - timedelta(days=days_since_monday + 7)
        last_sunday = last_monday + timedelta(days=6)
        
        logger.info(f"Generating scheduled weekly report for {last_monday} to {last_sunday}")
        
        result = await submit_weekly_report(
            start_date=last_monday.isoformat(),
            end_date=last_sunday.isoformat(),
            operator="system",
            run_id=str(uuid4())
        )
        
        logger.info(f"Weekly report generated: {result}")
        return result
    
    return asyncio.run(_run())
```
